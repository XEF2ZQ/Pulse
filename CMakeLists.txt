cmake_minimum_required(VERSION 3.22)
project(Pulse VERSION 1.2.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Pulse requires Windows and MSVC with a recent Windows 11 SDK.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Configure Pulse for x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(pulse_topology STATIC src/scheduler_topology.cpp)
target_include_directories(pulse_topology PUBLIC src)
target_compile_options(pulse_topology PRIVATE /W4 /WX /permissive- /utf-8)
add_executable(Pulse WIN32 src/main.cpp src/power.cpp src/pulse.rc)
target_link_libraries(Pulse PRIVATE pulse_topology powrprof user32 gdi32 shell32 ole32 advapi32 comctl32 dwmapi wtsapi32 psapi)
target_compile_options(Pulse PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf)
target_link_options(Pulse PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/pulse.manifest)
include(CTest)
add_executable(PulseTailProbe tools/tail_probe.cpp)
target_link_libraries(PulseTailProbe PRIVATE setupapi powrprof ole32 psapi user32)
target_compile_options(PulseTailProbe PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf)
target_link_options(PulseTailProbe PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
if(BUILD_TESTING)
    add_executable(tail_tests tests/tail_tests.cpp)
    target_link_libraries(tail_tests PRIVATE setupapi powrprof)
    target_compile_options(tail_tests PRIVATE /W4 /WX /permissive- /utf-8)
    add_test(NAME tail_telemetry COMMAND tail_tests)
    set_tests_properties(tail_telemetry PROPERTIES LABELS "unit")
    add_executable(demand_tests tests/demand_tests.cpp)
    target_include_directories(demand_tests PRIVATE src)
    target_compile_options(demand_tests PRIVATE /W4 /WX /permissive- /utf-8)
    add_test(NAME productive_demand COMMAND demand_tests)
    set_tests_properties(productive_demand PROPERTIES LABELS "unit")
    add_executable(demand_probe tests/demand_probe.cpp)
    target_include_directories(demand_probe PRIVATE src)
    target_link_libraries(demand_probe PRIVATE user32 psapi)
    target_compile_options(demand_probe PRIVATE /W4 /WX /permissive- /utf-8)
    add_executable(policy_tests tests/policy_tests.cpp)
    target_include_directories(policy_tests PRIVATE src)
    add_test(NAME power_policy COMMAND policy_tests)
    add_executable(scheduler_tests tests/scheduler_tests.cpp)
    target_link_libraries(scheduler_tests PRIVATE pulse_topology)
    target_compile_options(scheduler_tests PRIVATE /W4 /WX /permissive- /utf-8)
    add_test(NAME scheduler_unit COMMAND scheduler_tests --unit-only)
    set_tests_properties(power_policy scheduler_unit PROPERTIES LABELS "unit")
    add_executable(pulse_topology_dump tests/topology_dump.cpp)
    target_link_libraries(pulse_topology_dump PRIVATE pulse_topology)
    add_test(NAME live_topology COMMAND pulse_topology_dump)
    add_test(NAME scheduler_live COMMAND scheduler_tests)
    add_test(NAME scheduler_gui COMMAND Pulse --scheduler-gui-test ${CMAKE_CURRENT_BINARY_DIR}/scheduler-gui-test.txt)
    set_tests_properties(live_topology scheduler_live scheduler_gui PROPERTIES LABELS "integration;read-only")
endif()
