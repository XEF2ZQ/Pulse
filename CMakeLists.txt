cmake_minimum_required(VERSION 3.22)
project(Pulse VERSION 1.4.3.2 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Pulse requires Windows and MSVC with a recent Windows 11 SDK.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Configure Pulse for x64.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(pulse_topology STATIC src/scheduler_topology.cpp)
target_include_directories(pulse_topology PUBLIC src)
target_compile_options(pulse_topology PRIVATE /W4 /WX /permissive- /utf-8)
add_executable(Pulse WIN32 src/main.cpp src/power.cpp src/placement.cpp src/efficiency.cpp src/pcie_guard.cpp src/power_source.cpp src/pulse.rc)
target_link_libraries(Pulse PRIVATE pulse_topology powrprof user32 gdi32 shell32 ole32 oleaut32 wbemuuid wevtapi taskschd advapi32 comctl32 dwmapi wtsapi32 psapi uxtheme)
target_compile_options(Pulse PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf)
target_link_options(Pulse PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/pulse.manifest)
include(CTest)
add_executable(PulseGpuProbe tools/gpu_capabilities.cpp)
target_link_libraries(PulseGpuProbe PRIVATE user32 powrprof)
target_compile_options(PulseGpuProbe PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf)
target_link_options(PulseGpuProbe PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
set(PULSE_ADLX_SDK_ROOT "" CACHE PATH "Optional official ADLX checkout (not bundled)")
if(PULSE_ADLX_SDK_ROOT)
    if(NOT EXISTS "${PULSE_ADLX_SDK_ROOT}/SDK/Include/ADLX.h")
        message(FATAL_ERROR "PULSE_ADLX_SDK_ROOT must point to the ADLX repository root")
    endif()
    target_include_directories(PulseGpuProbe SYSTEM PRIVATE "${PULSE_ADLX_SDK_ROOT}/SDK/Include")
    target_compile_definitions(PulseGpuProbe PRIVATE PULSE_ADLX_PROBE)
endif()
add_executable(PulseTailProbe tools/tail_probe.cpp)
add_executable(PulsePowerSourceProbe tools/power_source_probe.cpp)
target_link_libraries(PulsePowerSourceProbe PRIVATE setupapi powrprof)
target_compile_options(PulsePowerSourceProbe PRIVATE /W4 /WX /permissive- /utf-8)
target_link_libraries(PulseTailProbe PRIVATE setupapi powrprof ole32 psapi user32)
target_compile_options(PulseTailProbe PRIVATE /W4 /WX /permissive- /utf-8 /guard:cf)
target_link_options(PulseTailProbe PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
